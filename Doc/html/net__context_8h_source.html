502static inline void net_context_set_family(struct net_context *context,
503 sa_family_t family)
504{
505 uint8_t flag = 0U;
506
507 NET_ASSERT(context);
508
509 if (family == AF_UNSPEC || family == AF_INET || family == AF_INET6 ||
510 family == AF_PACKET || family == AF_CAN) {
511 /* Family is in BIT(4), BIT(5) and BIT(6) */
512 flag = family << 3;
513 }
514
515 context->flags |= flag;
516}
