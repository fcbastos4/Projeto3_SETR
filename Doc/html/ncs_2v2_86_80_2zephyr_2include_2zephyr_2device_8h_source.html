424static inline device_handle_t device_handle_get(const struct device *dev)
425{
426 device_handle_t ret = DEVICE_HANDLE_NULL;
427 STRUCT_SECTION_START_EXTERN(device);
428
429 /* TODO: If/when devices can be constructed that are not part of the
430 * fixed sequence we'll need another solution.
431 */
432 if (dev != NULL) {
433 ret = 1 + (device_handle_t)(dev - STRUCT_SECTION_START(device));
434 }
435
436 return ret;
437}

447static inline const struct device *
448device_from_handle(device_handle_t dev_handle)
449{
450 STRUCT_SECTION_START_EXTERN(device);
451 const struct device *dev = NULL;
452 size_t numdev;
453
454 STRUCT_SECTION_COUNT(device, &numdev);
455
456 if ((dev_handle > 0) && ((size_t)dev_handle <= numdev)) {
457 dev = &STRUCT_SECTION_START(device)[dev_handle - 1];
458 }
459
460 return dev;
461}

504static inline const device_handle_t *
505device_required_handles_get(const struct device *dev, size_t *count)
506{
507 const device_handle_t *rv = dev->deps;
508
509 if (rv != NULL) {
510 size_t i = 0;
511
512 while ((rv[i] != Z_DEVICE_DEPS_ENDS) &&
513 (rv[i] != Z_DEVICE_DEPS_SEP)) {
514 ++i;
515 }
516 *count = i;
517 }
518
519 return rv;

540static inline const device_handle_t *
541device_injected_handles_get(const struct device *dev, size_t *count)
542{
543 const device_handle_t *rv = dev->deps;
544 size_t region = 0;
545 size_t i = 0;
546
547 if (rv != NULL) {
548 /* Fast forward to injected devices */
549 while (region != 1) {
550 if (*rv == Z_DEVICE_DEPS_SEP) {
551 region++;
552 }
553 rv++;
554 }
555 while ((rv[i] != Z_DEVICE_DEPS_ENDS) &&
556 (rv[i] != Z_DEVICE_DEPS_SEP)) {
557 ++i;
558 }
559 *count = i;
560 }
561
562 return rv;
563}

584static inline const device_handle_t *
585device_supported_handles_get(const struct device *dev, size_t *count)
586{
587 const device_handle_t *rv = dev->deps;
588 size_t region = 0;
589 size_t i = 0;
590
591 if (rv != NULL) {
592 /* Fast forward to supporting devices */
593 while (region != 2) {
594 if (*rv == Z_DEVICE_DEPS_SEP) {
595 region++;
596 }
597 rv++;
598 }
599 /* Count supporting devices.
600 * Trailing NULL's can be injected by gen_device_deps.py due to
601 * CONFIG_PM_DEVICE_POWER_DOMAIN_DYNAMIC_NUM
602 */
603 while ((rv[i] != Z_DEVICE_DEPS_ENDS) &&
604 (rv[i] != DEVICE_HANDLE_NULL)) {
605 ++i;
606 }
607 *count = i;
608 }
609
610 return rv;
611}
