59static inline void ring_buf_internal_reset(struct ring_buf *buf, int32_t value)
60{
61 buf->put_head = buf->put_tail = buf->put_base = value;
62 buf->get_head = buf->get_tail = buf->get_base = value;

157static inline void ring_buf_init(struct ring_buf *buf,
158 uint32_t size,
159 uint8_t *data)
160{
161 __ASSERT(size < RING_BUFFER_MAX_SIZE, RING_BUFFER_SIZE_ASSERT_MSG);
162
163 buf->size = size;
164 buf->buffer = data;
165 ring_buf_internal_reset(buf, 0);
166}
