324/* COND_CODE_1 is used to omit an LTV entry in case the _frames_per_sdu is 1.
325 * COND_CODE_1 will evaluate to second argument if the flag parameter(first argument) is 1
326 * - removing one layer of paranteses.
327 * If the flags argument is != 1 it will evaluate to the third argument which inserts a LTV
328 * entry for the max_frames_per_sdu value.

331#define BT_AUDIO_CODEC_CAP_LC3_DATA(_freq, _duration, _chan_count, _len_min, _len_max, \
332 _max_frames_per_sdu) \
333 { \
334 BT_AUDIO_CODEC_DATA(BT_AUDIO_CODEC_LC3_FREQ, BT_BYTES_LIST_LE16(_freq)), \
335 BT_AUDIO_CODEC_DATA(BT_AUDIO_CODEC_LC3_DURATION, (_duration)), \
336 BT_AUDIO_CODEC_DATA(BT_AUDIO_CODEC_LC3_CHAN_COUNT, (_chan_count)), \
337 BT_AUDIO_CODEC_DATA(BT_AUDIO_CODEC_LC3_FRAME_LEN, \
338 BT_BYTES_LIST_LE16(_len_min), \
339 BT_BYTES_LIST_LE16(_len_max)), \
340 COND_CODE_1(_max_frames_per_sdu, (), \
341 (BT_AUDIO_CODEC_DATA(BT_AUDIO_CODEC_LC3_FRAME_COUNT, \
342 (_max_frames_per_sdu)))) \
343 }
