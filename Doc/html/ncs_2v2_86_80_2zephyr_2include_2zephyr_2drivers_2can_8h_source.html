1590static inline uint8_t can_bytes_to_dlc(uint8_t num_bytes)
1591{
1592 return num_bytes <= 8 ? num_bytes :
1593 num_bytes <= 12 ? 9 :
1594 num_bytes <= 16 ? 10 :

1597 num_bytes <= 32 ? 13 :

1609static inline bool can_frame_matches_filter(const struct can_frame *frame,
1610 const struct can_filter *filter)
1611{
1612 if ((frame->flags & CAN_FRAME_IDE) != 0 && (filter->flags & CAN_FILTER_IDE) == 0) {
1613 /* Extended (29-bit) ID frame, standard (11-bit) filter */
1614 return false;
1615 }
1616
1617 if ((frame->flags & CAN_FRAME_IDE) == 0 && (filter->flags & CAN_FILTER_IDE) != 0) {
1618 /* Standard (11-bit) ID frame, extended (29-bit) filter */
1619 return false;
1620 }
1621
1622 if ((frame->flags & CAN_FRAME_RTR) == 0 && (filter->flags & CAN_FILTER_DATA) == 0) {
1623 /* non-RTR frame, remote transmission request (RTR) filter */
1624 return false;
1625 }
1626
1627 if ((frame->flags & CAN_FRAME_RTR) != 0 && (filter->flags & CAN_FILTER_RTR) == 0) {
1628 /* Remote transmission request (RTR) frame, non-RTR filter */
1629 return false;
1630 }
1631
1632 if ((frame->flags & CAN_FRAME_FDF) != 0 && (filter->flags & CAN_FILTER_FDF) == 0) {
1633 /* CAN FD format frame, classic format filter */
1634 return false;
1635 }
1636
1637 if ((frame->flags & CAN_FRAME_FDF) == 0 && (filter->flags & CAN_FILTER_FDF) != 0) {
1638 /* Classic frame, CAN FD format filter */
1639 return false;
1640 }
1641
1642 if ((frame->id ^ filter->id) & filter->mask) {
1643 /* Masked ID mismatch */
1644 return false;
1645 }
1646
1647 return true;
1648}
