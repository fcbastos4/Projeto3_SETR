400static inline int64_t arithmetic_shift_right(int64_t value, uint8_t shift)
401{
402 int64_t sign_ext;
403
404 if (shift == 0U) {
405 return value;
406 }
407
408 /* extract sign bit */
409 sign_ext = (value >> 63) & 1;
410
411 /* make all bits of sign_ext be the same as the value's sign bit */
412 sign_ext = -sign_ext;
413
414 /* shift value and fill opened bit positions with sign bit */
415 return (value >> shift) | (sign_ext << (64 - shift));
416}

446static inline void byteswp(void *a, void *b, size_t size)
447{
448 uint8_t t;
449 uint8_t *aa = (uint8_t *)a;
450 uint8_t *bb = (uint8_t *)b;
