47static inline void
48current_sense_amplifier_scale_dt(const struct current_sense_amplifier_dt_spec *spec,
49 int32_t *v_to_i)
50{
51 /* store in a temporary 64 bit variable to prevent overflow during calculation */
52 int64_t tmp = *v_to_i;
53
54 /* multiplies by 1,000,000 before dividing by sense resistance in micro-ohms. */
55 tmp = tmp * 1000000 / spec->sense_micro_ohms * spec->sense_gain_div / spec->sense_gain_mult;
56
57 *v_to_i = (int32_t)tmp;
58}
