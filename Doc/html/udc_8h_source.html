431static inline int udc_set_address(const struct device *dev, const uint8_t addr)
432{
433 const struct udc_api *api = dev->api;
434 int ret;
435
436 if (!udc_is_enabled(dev)) {
437 return -EPERM;
438 }
439
440 api->lock(dev);
441 ret = api->set_address(dev, addr);
442 api->unlock(dev);
443
444 return ret;
445}
