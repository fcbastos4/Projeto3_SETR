1357static inline void net_buf_destroy(struct net_buf *buf)
1358{
1359 struct net_buf_pool *pool = net_buf_pool_get(buf->pool_id);
1360
1361 if (buf->__buf) {
1362 if (!(buf->flags & NET_BUF_EXTERNAL_DATA)) {
1363 pool->alloc->cb->unref(buf, buf->__buf);
1364 }
1365 buf->__buf = NULL;
1366 }
1367
1368 k_lifo_put(&pool->free, buf);
1369}
