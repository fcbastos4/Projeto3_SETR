30static inline unsigned int irq_get_level(unsigned int irq)
31{
32 const uint32_t mask2 = BIT_MASK(CONFIG_2ND_LEVEL_INTERRUPT_BITS) <<
33 CONFIG_1ST_LEVEL_INTERRUPT_BITS;
34 const uint32_t mask3 = BIT_MASK(CONFIG_3RD_LEVEL_INTERRUPT_BITS) <<
35 (CONFIG_1ST_LEVEL_INTERRUPT_BITS + CONFIG_2ND_LEVEL_INTERRUPT_BITS);
36
37 if (IS_ENABLED(CONFIG_3RD_LEVEL_INTERRUPTS) && (irq & mask3) != 0) {
38 return 3;
39 }
40
41 if (IS_ENABLED(CONFIG_2ND_LEVEL_INTERRUPTS) && (irq & mask2) != 0) {
42 return 2;
43 }
44
45 return 1;
46}

58static inline unsigned int irq_from_level_2(unsigned int irq)
59{
60 if (IS_ENABLED(CONFIG_3RD_LEVEL_INTERRUPTS)) {
61 return ((irq >> CONFIG_1ST_LEVEL_INTERRUPT_BITS) &
62 BIT_MASK(CONFIG_2ND_LEVEL_INTERRUPT_BITS)) - 1;
63 } else {
64 return (irq >> CONFIG_1ST_LEVEL_INTERRUPT_BITS) - 1;
65 }
66}
