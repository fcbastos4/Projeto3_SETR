661static inline bool net_ipv6_is_prefix(const uint8_t *addr1,
662 const uint8_t *addr2,
663 uint8_t length)
664{
665 uint8_t bits = 128 - length;
666 uint8_t bytes = length / 8U;
667 uint8_t remain = bits % 8;
668 uint8_t mask;
669
670 if (length > 128) {
671 return false;
672 }
673
674 if (memcmp(addr1, addr2, bytes)) {
675 return false;
676 }
677
678 if (!remain) {
679 /* No remaining bits, the prefixes are the same as first
680 * bytes are the same.
681 */
682 return true;
683 }
684
685 /* Create a mask that has remaining most significant bits set */
686 mask = ((0xff << (8 - remain)) ^ 0xff) << remain;
687
688 return (addr1[bytes] & mask) == (addr2[bytes] & mask);
689}

1281static inline void net_ipv6_addr_create_iid(struct in6_addr *addr,
1282 struct net_linkaddr *lladdr)
1283{
1284 UNALIGNED_PUT(htonl(0xfe800000), &addr->s6_addr32[0]);
1285 UNALIGNED_PUT(0, &addr->s6_addr32[1]);
1286
1287 switch (lladdr->len) {
1288 case 2:
1289 /* The generated IPv6 shall not toggle the
1290 * Universal/Local bit. RFC 6282 ch 3.2.2
1291 */
1292 if (lladdr->type == NET_LINK_IEEE802154) {
1293 UNALIGNED_PUT(0, &addr->s6_addr32[2]);
1294 addr->s6_addr[11] = 0xff;
1295 addr->s6_addr[12] = 0xfe;
1296 addr->s6_addr[13] = 0U;
1297 addr->s6_addr[14] = lladdr->addr[0];
1298 addr->s6_addr[15] = lladdr->addr[1];
1299 }
1300
1301 break;
1302 case 6:
1303 /* We do not toggle the Universal/Local bit
1304 * in Bluetooth. See RFC 7668 ch 3.2.2
1305 */
1306 memcpy(&addr->s6_addr[8], lladdr->addr, 3);
1307 addr->s6_addr[11] = 0xff;
1308 addr->s6_addr[12] = 0xfe;
1309 memcpy(&addr->s6_addr[13], lladdr->addr + 3, 3);
1310
1311#if defined(CONFIG_NET_L2_BT_ZEP1656)
1312 /* Workaround against older Linux kernel BT IPSP code.
1313 * This will be removed eventually.
1314 */
1315 if (lladdr->type == NET_LINK_BLUETOOTH) {
1316 addr->s6_addr[8] ^= 0x02;
1317 }
1318#endif
1319
1320 if (lladdr->type == NET_LINK_ETHERNET) {
1321 addr->s6_addr[8] ^= 0x02;
1322 }
1323
1324 break;
1325 case 8:
1326 memcpy(&addr->s6_addr[8], lladdr->addr, lladdr->len);
1327 addr->s6_addr[8] ^= 0x02;
1328 break;
1329 }
1330}

1337static inline bool net_ipv6_addr_based_on_ll(const struct in6_addr *addr,
1338 const struct net_linkaddr *lladdr)
1339{
1340 if (!addr || !lladdr) {
1341 return false;
1342 }
1343
1344 switch (lladdr->len) {
1345 case 2:
1346 if (!memcmp(&addr->s6_addr[14], lladdr->addr, lladdr->len) &&
1347 addr->s6_addr[8] == 0U &&
1348 addr->s6_addr[9] == 0U &&
1349 addr->s6_addr[10] == 0U &&
1350 addr->s6_addr[11] == 0xff &&
1351 addr->s6_addr[12] == 0xfe) {
1352 return true;
1353 }
1354
1355 break;
1356 case 6:
1357 if (lladdr->type == NET_LINK_ETHERNET) {
1358 if (!memcmp(&addr->s6_addr[9], &lladdr->addr[1], 2) &&
1359 !memcmp(&addr->s6_addr[13], &lladdr->addr[3], 3) &&
1360 addr->s6_addr[11] == 0xff &&
1361 addr->s6_addr[12] == 0xfe &&
1362 (addr->s6_addr[8] ^ 0x02) == lladdr->addr[0]) {
1363 return true;
1364 }
1365 } else if (lladdr->type == NET_LINK_BLUETOOTH) {
1366 if (!memcmp(&addr->s6_addr[9], &lladdr->addr[1], 2) &&
1367 !memcmp(&addr->s6_addr[13], &lladdr->addr[3], 3) &&
1368 addr->s6_addr[11] == 0xff &&
1369 addr->s6_addr[12] == 0xfe
1370#if defined(CONFIG_NET_L2_BT_ZEP1656)
1371 /* Workaround against older Linux kernel BT IPSP
1372 * code. This will be removed eventually.
1373 */
1374 && (addr->s6_addr[8] ^ 0x02) == lladdr->addr[0]
1375#endif
1376 ) {
1377 return true;
1378 }
1379 }
1380
1381 break;
1382 case 8:
1383 if (!memcmp(&addr->s6_addr[9], &lladdr->addr[1],
1384 lladdr->len - 1) &&
1385 (addr->s6_addr[8] ^ 0x02) == lladdr->addr[0]) {
1386 return true;
1387 }
1388
1389 break;
1390 }
1391
1392 return false;
1393}
