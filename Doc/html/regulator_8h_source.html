174 .startup_delay_us = DT_PROP_OR(node_id, startup_delay_us, 0), \
175 .off_on_delay_us = DT_PROP_OR(node_id, off_on_delay_us, 0), \
176 .allowed_modes = (const regulator_mode_t []) \
177 DT_PROP_OR(node_id, regulator_allowed_modes, {}), \
178 .allowed_modes_cnt = \
179 DT_PROP_LEN_OR(node_id, regulator_allowed_modes, 0), \
180 .initial_mode = DT_PROP_OR(node_id, regulator_initial_mode, \
181 REGULATOR_INITIAL_MODE_UNKNOWN), \
182 .flags = ((DT_PROP_OR(node_id, regulator_always_on, 0U) * \

185 REGULATOR_BOOT_ON)), \

266static inline int regulator_common_get_min_voltage(const struct device *dev, int32_t *min_uv)
267{
268 const struct regulator_common_config *config =
