71 * list function are stored on the stack at addresses which are 32-bit
72 * aligned. It means that variables of type unit64_t and double may not

324do { \
325 BUILD_ASSERT(!((sizeof(double) < VA_STACK_ALIGN(long double)) && \
326 Z_CBPRINTF_IS_LONGDOUBLE(_arg) && \
327 !IS_ENABLED(CONFIG_CBPRINTF_PACKAGE_LONGDOUBLE)),\
328 "Packaging of long double not enabled in Kconfig."); \
329 while (_align_offset % Z_CBPRINTF_ALIGNMENT(_arg) != 0UL) { \
330 _idx += sizeof(int); \
331 _align_offset += sizeof(int); \
332 } \
333 uint32_t _arg_size = Z_CBPRINTF_ARG_SIZE(_arg); \
334 uint32_t _loc = _idx / sizeof(int); \
335 if (arg_idx < 1 + _fros_cnt) { \
336 if (_ros_pos_en) { \
337 _ros_pos_buf[_ros_pos_idx++] = _loc; \
338 } \
339 } else if (Z_CBPRINTF_IS_PCHAR(_arg, 0)) { \
340 if (_cros_en) { \
341 if (Z_CBPRINTF_IS_X_PCHAR(arg_idx, _arg, _flags)) { \
342 if (_rws_pos_en) { \
343 _rws_buffer[_rws_pos_idx++] = arg_idx - 1; \
344 _rws_buffer[_rws_pos_idx++] = _loc; \
345 } \
346 } else { \
347 if (_ros_pos_en) { \
348 _ros_pos_buf[_ros_pos_idx++] = _loc; \
349 } \
350 } \
351 } else if (_rws_pos_en) { \
352 _rws_buffer[_rws_pos_idx++] = arg_idx - 1; \
353 _rws_buffer[_rws_pos_idx++] = _idx / sizeof(int); \
354 } \
355 } \
356 if (_buf && _idx < (int)_max) { \
357 Z_CBPRINTF_STORE_ARG(&_buf[_idx], _arg); \
358 } \
359 _idx += _arg_size; \
360 _align_offset += _arg_size; \
361} while (false)

384/* Allocation to avoid using VLA and alloca. Alloc frees space when leaving
385 * a function which can lead to increased stack usage if logging is used
386 * multiple times. VLA is not always available.
387 *
388 * Use large array when optimization is off to avoid increased stack usage.
389 */
390#ifdef CONFIG_NO_OPTIMIZATIONS
391#define Z_CBPRINTF_ON_STACK_ALLOC(_name, _len) \
392 __ASSERT(_len <= 32, "Too many string arguments."); \
393 uint8_t _name##_buf32[32]; \
394 _name = _name##_buf32
395#else
396#define Z_CBPRINTF_ON_STACK_ALLOC(_name, _len) \
397 __ASSERT(_len <= 32, "Too many string arguments."); \
398 uint8_t _name##_buf4[4]; \
399 uint8_t _name##_buf8[8]; \
400 uint8_t _name##_buf12[12]; \
401 uint8_t _name##_buf16[16]; \
402 uint8_t _name##_buf32[32]; \
403 _name = (_len) <= 4 ? _name##_buf4 : \
404 ((_len) <= 8 ? _name##_buf8 : \
405 ((_len) <= 12 ? _name##_buf12 : \
406 ((_len) <= 16 ? _name##_buf16 : \
407 _name##_buf32)))
408#endif

410/* When the first argument of Z_CBPRINTF_STATIC_PACKAGE_GENERIC() is a
411 * static memory location, some compiler warns you if you compare the
412 * location against NULL. ___is_null() is used to kill this warning.
413 *
414 * The warnings would be visible when you built with -save-temps=obj,
415 * our standard debugging tip for macro problems.
416 *
417 * https://github.com/zephyrproject-rtos/zephyr/issues/51528
418 */
419static ALWAYS_INLINE bool ___is_null(void *p)
420{
421 return p == NULL;
422}

439#define Z_CBPRINTF_STATIC_PACKAGE_GENERIC(buf, _inlen, _outlen, _align_offset, \
440 flags, ... /* fmt, ... */) \
441do { \
442 _Pragma("GCC diagnostic push") \
443 _Pragma("GCC diagnostic ignored \"-Wpointer-arith\"") \
444 Z_CBPRINTF_SUPPRESS_SIZEOF_ARRAY_DECAY \
445 BUILD_ASSERT(!IS_ENABLED(CONFIG_XTENSA) || \
446 (IS_ENABLED(CONFIG_XTENSA) && \
447 !(_align_offset % CBPRINTF_PACKAGE_ALIGNMENT)), \
448 "Xtensa requires aligned package."); \
449 BUILD_ASSERT((_align_offset % sizeof(int)) == 0, \
450 "Alignment offset must be multiply of a word."); \
451 IF_ENABLED(CONFIG_CBPRINTF_STATIC_PACKAGE_CHECK_ALIGNMENT, \
452 (__ASSERT(!((uintptr_t)buf & (CBPRINTF_PACKAGE_ALIGNMENT - 1)), \
453 "Buffer must be aligned.");)) \
454 uint32_t _flags = flags; \
455 bool _ros_pos_en = (_flags) & CBPRINTF_PACKAGE_ADD_RO_STR_POS; \
456 bool _rws_pos_en = (_flags) & CBPRINTF_PACKAGE_ADD_RW_STR_POS; \
457 bool _cros_en = (_flags) & CBPRINTF_PACKAGE_CONST_CHAR_RO; \
458 uint8_t *_pbuf = buf; \
459 uint8_t _rws_pos_idx = 0; \
460 uint8_t _ros_pos_idx = 0; \
461 /* Variable holds count of all string pointer arguments. */ \
462 uint8_t _alls_cnt = Z_CBPRINTF_PCHAR_COUNT(0, __VA_ARGS__); \
463 uint8_t _fros_cnt = Z_CBPRINTF_PACKAGE_FIRST_RO_STR_CNT_GET(_flags); \
464 /* Variable holds count of non const string pointers. */ \
465 uint8_t _rws_cnt = _cros_en ? \
466 Z_CBPRINTF_PCHAR_COUNT(_flags, __VA_ARGS__) : _alls_cnt - _fros_cnt; \
467 uint8_t _ros_cnt = _ros_pos_en ? (1 + _alls_cnt - _rws_cnt) : 0; \
468 uint8_t *_ros_pos_buf; \
469 Z_CBPRINTF_ON_STACK_ALLOC(_ros_pos_buf, _ros_cnt); \
470 uint8_t *_rws_buffer; \
471 Z_CBPRINTF_ON_STACK_ALLOC(_rws_buffer, 2 * _rws_cnt); \
472 size_t _pmax = !___is_null(buf) ? _inlen : INT32_MAX; \
473 int _pkg_len = 0; \
474 int _total_len = 0; \
475 int _pkg_offset = _align_offset; \
476 union cbprintf_package_hdr *_len_loc; \
477 /* If string has rw string arguments CBPRINTF_PACKAGE_ADD_RW_STR_POS is a must. */ \
478 if (_rws_cnt && !((_flags) & CBPRINTF_PACKAGE_ADD_RW_STR_POS)) { \
479 _outlen = -EINVAL; \
480 break; \
481 } \
482 /* package starts with string address and field with length */ \
483 if (_pmax < sizeof(*_len_loc)) { \
484 _outlen = -ENOSPC; \
485 break; \
486 } \
487 _len_loc = (union cbprintf_package_hdr *)_pbuf; \
488 _pkg_len += sizeof(*_len_loc); \
489 _pkg_offset += sizeof(*_len_loc); \
490 /* Pack remaining arguments */\
491 FOR_EACH_IDX(Z_CBPRINTF_PACK_ARG, (;), __VA_ARGS__);\
492 _total_len = _pkg_len; \
493 /* Append string indexes to the package. */ \
494 _total_len += _ros_cnt; \
495 _total_len += 2 * _rws_cnt; \
496 if (_pbuf != NULL) { \
497 /* Append string locations. */ \
498 uint8_t *_pbuf_loc = &_pbuf[_pkg_len]; \
499 for (size_t _ros_idx = 0; _ros_idx < _ros_cnt; _ros_idx++) { \
500 *_pbuf_loc++ = _ros_pos_buf[_ros_idx]; \
501 } \
502 for (size_t _rws_idx = 0; _rws_idx < (2 * _rws_cnt); _rws_idx++) { \
503 *_pbuf_loc++ = _rws_buffer[_rws_idx]; \
504 } \
505 } \
506 /* Store length */ \
507 _outlen = (_total_len > (int)_pmax) ? -ENOSPC : _total_len; \
508 /* Store length in the header, set number of dumped strings to 0 */ \
509 if (_pbuf != NULL) { \
510 union cbprintf_package_hdr pkg_hdr = { \
511 .desc = { \
512 .len = (uint8_t)(_pkg_len / sizeof(int)), \
513 .str_cnt = 0, \
514 .ro_str_cnt = _ros_cnt, \
515 .rw_str_cnt = _rws_cnt, \
516 } \
517 }; \
518 IF_ENABLED(CONFIG_CBPRINTF_PACKAGE_HEADER_STORE_CREATION_FLAGS, \
519 (pkg_hdr.desc.pkg_flags = flags)); \
520 *_len_loc = pkg_hdr; \
521 } \
522 _Pragma("GCC diagnostic pop") \
523} while (false)
