312static inline uint32_t z_impl_counter_us_to_ticks(const struct device *dev,
313 uint64_t us)
314{
315 uint64_t ticks = (us * z_impl_counter_get_frequency(dev)) / USEC_PER_SEC;
316
317 return (ticks > (uint64_t)UINT32_MAX) ? UINT32_MAX : ticks;
